# 9-1. Keyframe index change
# ================================
# Calculate decrease point
df <- df %>%
  arrange(no) %>%
  mutate(
    keyframe_diff = best_keyframe_list_idx - lag(best_keyframe_list_idx),
    decrease_flag = keyframe_diff < 0
  )

# Decrease count
decrease_count <- sum(df$decrease_flag, na.rm = TRUE)

# Decrease ratio
total_steps <- sum(!is.na(df$keyframe_diff))
decrease_ratio <- decrease_count / total_steps

cat("Number of decreases in best_keyframe_list_idx:", decrease_count, "\n")
## Number of decreases in best_keyframe_list_idx: 186
cat("Decrease ratio:", round(decrease_ratio, 3), "\n")
## Decrease ratio: 0.031
# Graph
p9_keyframe <- ggplot(df, aes(x = no, y = best_keyframe_list_idx)) +

# ================================
# 9-2. Matching metrics + inlier ratio
#      Left axis : counts
#      Right axis: inlier_ratio
# ================================
count_cols <- c("num_matches", "num_inliers", "num_outliers")

count_max <- max(unlist(df[count_cols]), na.rm = TRUE)
ratio_max <- max(df$inlier_ratio, na.rm = TRUE)

if (is.na(ratio_max) || ratio_max == 0) {
  ratio_scale_factor <- 1
} else {
  ratio_scale_factor <- count_max / ratio_max
}

counts_long <- df %>%
  select(no, all_of(count_cols)) %>%
  pivot_longer(
    cols = all_of(count_cols),
    names_to = "metric",
    values_to = "value"
  )

p9_metrics <- ggplot() +
  geom_line(
    data = counts_long,
    aes(x = no, y = value, color = metric),
    linewidth = 0.05
  ) +
  geom_point(
    data = counts_long,
    aes(x = no, y = value, color = metric),
    size = 0.1
  ) +
  geom_line(
    data = df,
    aes(x = no, y = inlier_ratio * ratio_scale_factor, color = "inlier_ratio"),
    linewidth = 0.05,
    linetype = "dashed"
  ) +
  geom_point(
    data = df,
    aes(x = no, y = inlier_ratio * ratio_scale_factor, color = "inlier_ratio"),
    size = 0.1
  ) +
  scale_y_continuous(
    name = "Counts (num_matches / num_inliers / num_outliers)",
    sec.axis = sec_axis(
      trans = ~ . / ratio_scale_factor,
      name = "inlier_ratio"
    )
  ) +
  ggtitle("Matching Metrics and Inlier Ratio vs Test Image Number") +
  xlab("no") +
  labs(color = "Metric") +
  theme_minimal()
